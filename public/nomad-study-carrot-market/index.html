use loading:

const Bs = dynamic(() => import("@components/bs"), {

use Suspense

import dynamic from "next/dynamic";

const Bs = dynamic(() => import("@components/bs"));
const Bs = dynamic(() => import("@components/bs"), {
  ssr: false,
  suspense: true,
});

<Suspense fallback={"Loading something big"}>
  <Bs />
</Suspense>;
